using System.Security.Cryptography;
using NetDropbox.Application.Abstractions;
using NetDropbox.Domain;

namespace NetDropbox.Infrastructure.Storage;

public sealed class FileSystemChunkStorage(string root) : IChunkStorage
{
    public async Task<StoredChunk> StoreAsync(Guid sessionId, int index, Stream content, CancellationToken cancellationToken)
    {
        var path = ChunkPath(sessionId, index);
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);

        await using var file = File.Create(path);
        using var sha = SHA256.Create();
        var buffer = new byte[1024 * 128];
        long total = 0;
        int read;
        while ((read = await content.ReadAsync(buffer, cancellationToken)) > 0)
        {
            await file.WriteAsync(buffer.AsMemory(0, read), cancellationToken);
            sha.TransformBlock(buffer, 0, read, null, 0);
            total += read;
        }
        sha.TransformFinalBlock([], 0, 0);
        return new StoredChunk(Convert.ToHexString(sha.Hash!).ToLowerInvariant(), total);
    }

    public async Task<Stream> OpenAssembledAsync(UploadSession session, IReadOnlyCollection<int> chunkIndexes, CancellationToken cancellationToken)
    {
        var temp = Path.Combine(root, $"{session.Id}.assembled");
        await using (var output = File.Create(temp))
        {
            foreach (var index in chunkIndexes.OrderBy(x => x))
            {
                await using var input = File.OpenRead(ChunkPath(session.Id, index));
                await input.CopyToAsync(output, cancellationToken);
            }
        }
        return File.Open(temp, FileMode.Open, FileAccess.Read, FileShare.Read);
    }

    public Task DeleteSessionAsync(Guid sessionId, CancellationToken cancellationToken)
    {
        var directory = SessionPath(sessionId);
        if (Directory.Exists(directory)) Directory.Delete(directory, recursive: true);
        var assembled = Path.Combine(root, $"{sessionId}.assembled");
        if (File.Exists(assembled)) File.Delete(assembled);
        return Task.CompletedTask;
    }

    private string SessionPath(Guid sessionId) => Path.Combine(root, sessionId.ToString("N"));
    private string ChunkPath(Guid sessionId, int index) => Path.Combine(SessionPath(sessionId), $"{index:D8}.chunk");
}
